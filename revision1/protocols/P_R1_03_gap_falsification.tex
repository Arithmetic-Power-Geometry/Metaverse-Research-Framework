\begin{algorithm}[H]\caption{Contrary-Evidence Gap Falsification Procedure}\label{proc:gap}\begin{algorithmic}[1]\Require Candidate gap $g$ with exact scope\State Search closest reviews and surveys\State Search evaluated primary studies\State Search datasets, benchmarks, testbeds, and artifacts\State Search final standards and active work separately when applicable\State Record explicit contrary evidence\If{direct contrary evidence satisfies the absence claim}\State \Return \textsc{Reject}\ElsIf{only a narrower evidence condition remains}\State \Return \textsc{Narrow/Retain}\Else\State \Return \textsc{Hold/Retain} with provenance\EndIf\State Stop only after saturated domain and applicable evidence classes have been checked\end{algorithmic}\end{algorithm}
